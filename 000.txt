def crear_estadistica(request):
    equipos = equipo.objects.all()

    if request.method == "POST":
        Estadisticas.objects.update_or_create(
            equipo_id=request.POST.get("equipo"),
            defaults={
                "jj": request.POST.get("jj"),
                "jg": request.POST.get("jg"),
                "jp": request.POST.get("jp"),
                "puntos": request.POST.get("puntos"),
                "gaf": request.POST.get("gaf"),
                "gc": request.POST.get("gc"),
                "dif": request.POST.get("dif"),
                "amonestaciones": request.POST.get("amon"),
            }
        )
        return redirect("tabla_posiciones")

    return render(request, "Resultados/crear_estadistica.html", {
        "equipos": equipos
    })
